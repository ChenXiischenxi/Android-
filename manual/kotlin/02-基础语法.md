```kotlin
fun main() {

    val a: Int = 5

    val b: Int = 10


    println(a == b)      // false，结构相等比较的是值

    println(a != b)      // true

    println(a < b)       // true

    println(a > b)       // false

    println(a <= b)      // true

    println(a >= b)      // false


    // Int 是基本类型，=== 退化为和 == 相同

    val x: Int = 1000

    val y: Int = 1000

    println(x == y)      // true

    println(x === y)     // true，对基本类型 === 等价于 ==


    // 要观察引用相等，必须使用装箱类型 Int?

    val p: Int? = 1000

    val q: Int? = 1000

    println(p == q)      // true，值相等

    println(p === q)     // false，1000 超出缓存区间，两个不同的装箱对象


    val m: Int? = 100

    val n: Int? = 100

    println(m === n)     // true，命中 JVM Integer 缓存 -128..127

}
```

```kotlin
fun main() {

    // arrayOf 根据元素推断类型，得到 Array<Int>

    val a = arrayOf(1, 2, 3)


    // Array(3) { ... } 是工厂函数：it 是下标，从 0 开始

    val b = Array(3) { it * 2 }        // [0, 2, 4]


    println(a[0])                      // 1

    println(b[1])                      // 2

    println(a.size)                    // 3

    println(b.contentToString())       // [0, 2, 4]


    // 方括号等价于调用 get 和 set

    a[0] = 100

    println(a[0])                      // 100

    println(a.contentToString())       // [100, 2, 3]


    // 基本类型数组避免装箱，性能更好

    val ints: IntArray = intArrayOf(1, 2, 3)

    ints[0] = ints[1] + ints[2]

    println(ints.contentToString())    // [5, 2, 3]

}
```

```kotlin
    val squares = IntArray(5) { it * it }   // it 是下标

}
```

```kotlin
    // 闭区间 1..3：包含左右两端

    for (i in 1..3) print("$i")

    println("  => 1..3")


    // until：包含左端，不包含右端

    for (i in 1 until 3) print("$i")

    println("  => 1 until 3")


    // downTo：从大到小递减

    for (i in 3 downTo 1) print("$i")

    println("  => 3 downTo 1")


    // step：指定步长，步长必须为正数

    for (i in 1..6 step 2) print("$i")

    println("  => 1..6 step 2")


    // ..< 半开区间，Kotlin 1.9 起稳定

    for (i in 1..<3) print("$i")

    println("  => 半开区间")

}
```

```kotlin
123  => 1..3
12  => 1 until 3
321  => 3 downTo 1
135  => 1..6 step 2
12  => 半开区间
```

```kotlin
fun main() {

    val sites = listOf("Runoob", "www.runoob.com", "RUNOOB")


    // indices 返回下标区间 0..2，适合需要按下标再取值的场景

    for (i in sites.indices) {

        println("下标 $i -> ${sites[i]}")

    }


    println("---")


    // withIndex() 同时给出下标和值，(index, value) 是解构声明

    for ((index, value) in sites.withIndex()) {

        println("$index: $value")

    }

}
```

```kotlin
下标 1 -> www.runoob.com
下标 2 -> RUNOOB
---
0: Runoob
1: www.runoob.com
2: RUNOOB
```

```kotlin
fun main() {

    val info = mapOf("name" to "Runoob", "site" to "www.runoob.com")


    for ((key, value) in info) {

        println("$key = $value")

    }

}
```

```kotlin
fun main() {

    // loop@ 是外层循环的标签

    loop@ for (i in 1..3) {

        for (j in 1..3) {

            if (j == 2) continue@loop   // 跳过外层循环的本次迭代

            if (i == 3) break@loop      // 直接结束外层循环

            println("i = $i, j = $j")

        }

    }

    println("循环结束")

}
```

```kotlin
i = 1, j = 1
i = 2, j = 1
循环结束
```

```kotlin
}
```

```kotlin
fun main() {

    val numbers = listOf(1, null, 3, null, 5)


    for (number in numbers) {

        // run 是内联函数，它 lambda 里的 continue 作用于外层的 for 循环

        val value = number ?: run {

            println("遇到 null，跳过本次循环")

            continue

        }

        println("处理数字：$value")

    }

    println("循环结束")

}
```

```kotlin
遇到 null，跳过本次循环
```

```kotlin
fun main() {

    for (number in numbers) {
```

```kotlin
fun main() {

    val numbers = listOf(1, 2, 0, 4)


    // return@forEach 只结束本次 lambda 调用，等价于跳过 0

    numbers.forEach {

        if (it == 0) return@forEach

        println("forEach：$it")

    }


    println("---")


    // 用 for 循环加 continue 也能达到同样效果

    for (number in numbers) {

        if (number == 0) continue

        println("for：$number")

    }

}
```

```kotlin
forEach：2
forEach：4
---
for：1

for：4
```
