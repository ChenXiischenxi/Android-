```kotlin
// 文件路径：Main.kt

class Outer(val name: String = "OUTER") {

    private val secret = "runoob"


    // 嵌套类：不持有外部类实例引用，访问不了 name 和 secret

    class Nested {

        fun hello(): String = "我是嵌套类，拿不到外部类的成员"

    }


    // 内部类：持有外部类实例引用，可以访问外部类成员

    inner class Inner {

        fun hello(): String = "我是内部类，name=$name，secret=$secret"


        // this@Outer 明确指向外部类实例

        fun outerName(): String = this@Outer.name

    }

}


fun main() {

    // 嵌套类直接用外部类名访问，不需要外部类对象

    val nested = Outer.Nested()

    println(nested.hello())


    // 内部类必须依赖外部类实例

    val outer = Outer()

    val inner = outer.Inner()

    println(inner.hello())

    println("this@Outer 拿到：${inner.outerName()}")

}
```

```kotlin
我是嵌套类，拿不到外部类的成员
我是内部类，name=OUTER，secret=runoob
this@Outer 拿到：OUTER
```

```kotlin
// 文件路径：Main.kt

class Site(val name: String, val url: String) {

    // 重写 toString，让打印结果可读

    override fun toString(): String = "Site(name=$name, url=$url)"


    // 重写 equals，改为按内容比较

    override fun equals(other: Any?): Boolean {

        if (this === other) return true

        if (other !is Site) return false

        return name == other.name && url == other.url

    }


    // 重写 equals 后必须一起重写 hashCode，保证相等对象哈希码相同

    override fun hashCode(): Int = 31 * name.hashCode() + url.hashCode()

}


fun main() {

    val a = Site("RUNOOB", "www.runoob.com")

    val b = Site("RUNOOB", "www.runoob.com")


    println(a)

    println("a == b：${a == b}")                     // 内容相等

    println("a === b：${a === b}")                   // 不是同一个对象

    println("哈希码相同：${a.hashCode() == b.hashCode()}")

}
```

```kotlin
a == b：true
a === b：false
哈希码相同：true
```

```kotlin
    val text_id: String = "102"
```

```kotlin
class ResourceLoader(id: ResourceID) {
    val d: ResourceID = id
    //表明该函数返回的值是ReadOnlyProperty<MyUI, String>
    //在实例化MyUI的时候，立即做这里的检查
    operator fun provideDelegate( thisRef: MyUI, prop: KProperty<*>): ReadOnlyProperty<MyUI, String> {
        if(checkProperty(thisRef, prop.name)){
            //匹配则返回相应的委托
            return DellImpl(d)

            //不匹配则抛出异常
            throw Exception("Error ${prop.name}")
        }

    private fun checkProperty(thisRef: MyUI, name: String):Boolean {
        return name.equals("image") || name.equals("text")
    }
}
```

```kotlin
import kotlin.math.abs


@JvmInline

value class Money(val cents: Long) {

    operator fun plus(other: Money) = Money(cents + other.cents)


    operator fun minus(other: Money) = Money(cents - other.cents)


    // 数量倍乘

    operator fun times(quantity: Int) = Money(cents * quantity)


    operator fun compareTo(other: Money): Int = cents.compareTo(other.cents)


    // 输出成 19.99 这样的金额格式

    override fun toString(): String {

        val sign = if (cents < 0) "-" else ""

        val absCents = abs(cents)

        return "$sign${absCents / 100}.${(absCents % 100).toString().padStart(2, '0')}"

    }

}


fun main() {
```
